version: '3.8'

services:
  mysql:
    image: mysql:8.0
    container_name: rag_mysql
    environment:
      MYSQL_ROOT_PASSWORD: password
      MYSQL_DATABASE: ragdb
      MYSQL_USER: raguser
      MYSQL_PASSWORD: ragpass
    volumes:
      - mysql_data:/var/lib/mysql
    ports:
      - "3306:3306"
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      timeout: 20s
      retries: 10

  app:
    build: .
    container_name: rag_app
    environment:
      MYSQL_HOST: mysql
      MYSQL_PORT: 3306
      MYSQL_USER: root
      MYSQL_PASSWORD: password
      MYSQL_DATABASE: ragdb
      LANCEDB_PATH: /app/lancedb
      OPENAI_API_KEY: ${OPENAI_API_KEY}
    volumes:
      - lancedb_data:/app/lancedb
    ports:
      - "8000:8000"
    depends_on:
      mysql:
        condition: service_healthy

volumes:
  mysql_data:
